\subsection{Request Lifecycle and Holds}\label{cases:lifecycle}
\tcarea{LIF}
These cases establish that a submitted Request moves to Allocated, Pending, or Waiting by the stated rule, all or nothing; that Holds block capacity and reserve credits only while the Objection Window runs; that allocation commits everything in one step; and that a Waiting Request is allocated only when all its Required lines can be satisfied at once.

%   Lifecycle.States

%   Lifecycle.Waiting.Reevaluate

\begin{testcase}{lif-direct}{A Request without Matched lines goes from Submitted straight to Allocated}{Lifecycle.Evaluate, Lifecycle.States, Lifecycle.Allocate}
\tcfield{Actor}{\code{alice} (Manager, Platform)}
\tcfield{Priority}{High}
\tcfield{Preconditions}{Test world.}
\tcfield{Test data}{L1: Room A, D 10:00--12:00 (Q1 = 30.00). L2: GPU, 4 units, D 10:00--12:00 (Q7 = 60.00).}
\tcsteps
\tcstep{Submit a Request A1 with L1 and L2.}{L1, L2}{A1 is Allocated; it never enters Pending.}
\tcstep{Open A1's history.}{None}{Created, Submitted, and Allocated in that order, each with actor Alice Chen and a timestamp; both lines Allocated.}
\tcstep{Open the Statement of Platform.}{None}{Charges of 30.00 and 60.00, each equal to its locked Quote; Balance 4910.00.}
\tcstep{Open the notifications.}{None}{A notification that A1 was Allocated.}
\tcfield{Pass criteria}{A1 and both lines are Allocated and charged exactly once.}
\tcfield{Automated by}{\code{RequestLifecycleTest}: ``Lifecycle.Allocate: every Required line fits, so the lines are allocated and charged their locked Quotes''; \code{SubmitEndToEndTest}}
\end{testcase}

\tcfield{Actor}{\code{alice} (Manager, Platform)}
\tcfield{Priority}{Medium}

\tcsteps
\tcstep{Submit A1 again through the API.}{A1}{Refused with \code{WRONG\_STATE}.}

\tcstep{Submit A3.}{A3}{Refused with \code{EMPTY\_REQUEST}.}
\tcstep{Open the history of A1 and A2.}{None}{Every state change is listed with its actor (a Member, or the system for timed events) and a timestamp.}

\tcfield{Automated by}{\code{RequestStateTest}; \code{RequestLifecycleTest}: ``WRONG\_STATE: \ldots'', ``EMPTY\_REQUEST: a Draft without lines cannot be submitted''}
\end{testcase}
