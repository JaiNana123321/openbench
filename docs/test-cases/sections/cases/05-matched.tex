\subsection{Matched Allocation}\label{cases:matched}
\tcarea{MAT}
These cases establish that the Shortlist contains exactly the Members of other Teams who have the skills, the availability, and no overlapping Loan, in the stated order; that the requesting Manager selects Candidates and each lending Manager decides; that refusals and silence are handled without restarting the Objection Window; and that an approved Candidate becomes a Loan with its Earning.

\begin{testcase}{mat-shortlist-build}{The Shortlist holds the qualified Members of other Teams}{Matched.Shortlist.Build, Lifecycle.Evaluate}
\tcfield{Actor}{\code{alice} (Manager, Platform)}
\tcfield{Priority}{High}
\tcfield{Preconditions}{Test world with the Matched data.}
\tcfield{Test data}{M1: Person, skill Statistics, quantity 1, days D to D+1, Required.}
\tcsteps
\tcstep{Create a Draft with M1 and submit it.}{M1}{The Request is Pending: M1 needs an approval.}
\tcstep{Open M1's Shortlist.}{None}{Dave Kim (Operations, Statistics, 0 person-days lent), then Rita Gomez (Research, Statistics and Microscopy, 0). Ben Ortiz is absent: he is in Platform and lacks Statistics. No Candidate is selected yet.}
\tcstep{Check what each row shows.}{None}{Name, Team, skills, and person-days lent only; no link to the Candidate's other Requests or Statements.}
\tcfield{Pass criteria}{The Shortlist has exactly two Candidates in the stated order, and nothing was selected automatically.}
\tcfield{Automated by}{None yet}
\end{testcase}

\begin{testcase}{mat-select}{The Manager selects exactly the line's quantity; a short Shortlist leaves the Request Waiting}{Matched.Shortlist.Build, Matched.Approve.Route}
\tcfield{Actor}{\code{bob} (Manager, Research); \code{alice} where named}
\tcfield{Priority}{High}
\tcfield{Preconditions}{Test world with the Matched data.}
\tcfield{Test data}{M2: Person, skill Statistics, quantity 2, days D to D+1, Required. M1: Person, skill Statistics, quantity 1, days D to D+1 (for \code{alice}).}
\tcsteps
\tcstep{Submit a Request with M2 and open its Shortlist.}{M2}{Dave Kim only, since Rita is in Research. The Request is Waiting: M2 is Required and has fewer Candidates than its quantity.}
\tcstep{As \code{alice}, submit a Request with M1 and try to select both Candidates.}{Dave Kim and Rita Gomez}{Refused with \code{LINE\_NOT\_ATOMIC}: exactly one Candidate must be selected.}
\tcstep{As \code{alice}, try to select a Member who is not on the Shortlist through the API.}{Carol Davis (Operations, no skills)}{Refused with \code{NOT\_A\_CANDIDATE}.}
\tcstep{As \code{alice}, select one Candidate.}{Dave Kim}{Dave is selected and an approval request is sent; Dave receives a notification that he was selected.}
\tcfield{Pass criteria}{No Candidate was selected without the Manager's action.}
\tcfield{Automated by}{None yet}
\end{testcase}

\begin{testcase}{mat-approve-route}{The approval request goes to the Candidate's Manager, who decides}{Matched.Approve.Route}

\tcsteps
\tcstep{Open Approvals.}{None}{One approval request: Candidate Dave Kim, requesting Team Platform, days D to D+1, Earning 800.00 (EA).}
\tcstep{As \code{alice}, try to decide the approval through the API.}{Approve}{Refused with \code{FORBIDDEN}: the requesting Manager cannot decide.}

\tcfield{Pass criteria}{Exactly one approval request was sent, to Oscar, and only his decision counted.}
\tcfield{Automated by}{None yet}
\end{testcase}

\begin{testcase}{mat-refuse-replace}{A refusal asks for a replacement without restarting the Objection Window}{Matched.Refuse.Replace, Lifecycle.Waiting.Reevaluate, Lifecycle.Window.Start}
\tcfield{Actor}{\code{alice} (Manager, Platform); \code{oscar} and \code{bob} where named}
\tcfield{Priority}{High}
\tcfield{Preconditions}{\code{alice} has a Pending Request A1 with M1 (Statistics, quantity 1, days D to D+1, Required, Dave Kim selected) and L1: Room A, D 10:00--12:00, on Hold (30.00 held). A1's Objection Window deadline is shown on A1, say W.}
\tcfield{Test data}{Reason ``Dave is on another project''.}
\tcsteps
\tcstep{As \code{oscar}, refuse the approval with a reason.}{Reason}{Recorded with actor, reason, and timestamp.}
\tcstep{Open A1.}{None}{A1 is Waiting; L1's Hold is released and the 30.00 is no longer held; A1 asks Alice to select a replacement for M1; the deadline is still W.}
\tcstep{Open M1's Shortlist and select a replacement.}{Rita Gomez}{A1 returns to Pending; L1 is on Hold again; an approval request goes to Bob Patel; the deadline is still W.}
\tcstep{As \code{bob}, approve.}{None}{A1 is Allocated; Rita has a Loan to Platform for D to D+1.}
\tcfield{Pass criteria}{The Objection Window never restarted, and Dave Kim has no Loan.}
\tcfield{Automated by}{None yet}
\end{testcase}
