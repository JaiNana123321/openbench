\subsection{Draft and Submit a Request}\label{cases:draft}
\tcarea{DRF}
These cases establish that only a Manager drafts, submits, and withdraws their own Team's Requests; that every line is checked as it is added and priced while drafting; that a Request the Team cannot afford is refused at submission with its shortfall; and that submission locks every Quote.

\begin{testcase}{drf-create}{A Manager creates a Draft for their own Team}{Draft.Create, Lifecycle.States}
\tcfield{Actor}{\code{alice} (Manager, Platform); \code{ben} where named}
\tcfield{Priority}{High}
\tcfield{Preconditions}{Test world.}
\tcfield{Test data}{None.}
\tcsteps
\tcstep{Open Requests and create a Request.}{None}{The Draft editor opens for a new Request of Team Platform in state Draft; its history shows ``Created'' by Alice Chen with a timestamp.}
\tcstep{Return to Requests.}{None}{The new Draft is listed with Team Platform and state Draft.}
\tcstep{Sign in as \code{ben} and look for Requests.}{None}{The menu has no Requests entry, and opening \code{/requests} shows the not-allowed page.}
\tcstep{As \code{ben}, create a Request through the API.}{POST \path{/api/requests}}{Refused with \code{FORBIDDEN}; no Request is created.}
\tcfield{Pass criteria}{Exactly one Request exists, a Draft of Platform created by Alice.}
\tcfield{Automated by}{\code{DraftRequestTest}: ``Draft.Create: a Manager's new Draft belongs to their Team and is recorded in its history''; \code{MembersControllerTest}: ``SE-roles: a caller without the role a service needs gets 403 FORBIDDEN''}
\end{testcase}

\begin{testcase}{drf-required-optional}{Lines are Required by default and may be split to make a part Optional}{Draft.Line.Required, Draft.Quote.Line}
\tcfield{Actor}{\code{alice} (Manager, Platform)}
\tcfield{Priority}{High}
\tcfield{Preconditions}{\code{alice} has an empty Draft.}
\tcfield{Test data}{L1: Meeting room, quantity 2, D 10:00--12:00 (Quote 60.00). L2: GPU, 6 units, D 10:00--12:00 (Q11 = 90.00).}
\tcsteps
\tcstep{Add L1 and L2.}{L1, L2}{Both Required; Required lines 150.00; Optional lines 0.00.}
\tcstep{Split L1, moving one room to a new line.}{1 of 2}{L1 has quantity 1 and Quote 30.00, Required; a new line L1b of the same type and window has quantity 1, Quote 30.00, Optional. Required 120.00; Optional 30.00.}
\tcstep{Split L2, moving 2 units.}{2 of 6}{L2 keeps 4 units (60.00), Required; L2b has 2 units (30.00), Optional. Required 90.00; Optional 60.00.}
\tcstep{Try to split L1 again, moving its only room.}{1 of 1}{Refused with \code{LINE\_NOT\_ATOMIC}: both parts must keep at least one.}
\tcstep{Mark L1b Required.}{Required}{Required 120.00; Optional 30.00.}
\tcfield{Pass criteria}{No line offers a partial-acceptance setting; only whole lines are Required or Optional.}
\tcfield{Automated by}{\code{DraftCasesTest}: ``drf-required-optional: Required by default; a split moves a part to an Optional line; marking it Required again''; \code{DraftRequestTest}: ``Draft.Line.Split: \ldots'' (both), ``Split refuses with 422 LINE\_NOT\_ATOMIC unless both parts keep at least 1''}
\end{testcase}

\begin{testcase}{drf-shortfall}{Submission refused while the Required sum exceeds the Available Balance}{Draft.Quote.Line, Ledger.Account}
\tcfield{Actor}{\code{bob} (Manager, Research)}
\tcfield{Priority}{High}
\tcfield{Preconditions}{Test world: Research's Available Balance is 600.00. \code{bob} has an empty Draft R1.}
\tcfield{Test data}{L1: Confocal microscope, quantity 1, D 09:00--17:00 (Q9 = 480.00), Required. L2: Confocal microscope, quantity 1, D+1 09:00--12:00 (Q10 = 180.00), Required.}
\tcsteps
\tcstep{Add L1 and L2.}{L1, L2}{Required lines 660.00; Available Balance 600.00; shortfall 60.00 shown.}
\tcstep{Submit R1.}{None}{Refused with \code{INSUFFICIENT\_BALANCE} and the message ``short by 60.00 credits''; R1 stays a Draft.}
\tcstep{Mark L2 Optional and submit R1.}{L2 Optional}{Required 480.00 is within 600.00, so submission is accepted; the Optional line does not count toward the refusal.}
\tcstep{Open the Statement of Research.}{None}{One Charge of 480.00 for L1. L2 was not affordable after L1 was charged, so it is Dropped and charged nothing (\TC{lif-optional-drop}).}
\tcfield{Pass criteria}{After step 2 the Statement and Inventory were unchanged; after step 4 Research's Balance is 120.00.}
\tcfield{Automated by}{\code{RequestLifecycleTest}: ``INSUFFICIENT\_BALANCE: a Request costing more than the Available Balance is refused with the shortfall \ldots''; \code{DraftRequestTest}: ``Draft.Quote.Totals: the shortfall is the Required sum above the Available Balance; Optional lines do not count''}
\end{testcase}

\begin{testcase}{drf-submit-lock}{Submission locks every Quote and records the submission time}{Draft.Submit.Lock, Planner.Settings}

\tcfield{Priority}{High}
\tcfield{Preconditions}{\code{alice} has a Draft R1 with L1: Room A, D 10:00--12:00 (30.00), and a Draft R2 with L2: Room B, D 14:00--16:00 (30.00).}
\tcfield{Test data}{New Unit Price for Meeting room: 20.00.}
\tcsteps

\begin{testcase}{drf-withdraw}{Withdrawing from Draft, Waiting, or Pending releases everything held}{Draft.Create, Lifecycle.Hold.Place}

\tcfield{Test data}{None.}
\tcsteps
\tcstep{Withdraw B1.}{None}{B1 is Withdrawn; editing or submitting it is refused with \code{WRONG\_STATE}.}
\tcstep{Withdraw B2.}{None}{B2 is Withdrawn; its line leaves the waitlist for Room A.}

\tcstep{Try to withdraw \code{alice}'s Allocated Request through the API as \code{alice}.}{Its id}{Refused with \code{WRONG\_STATE}: an Allocated Request is released line by line, not withdrawn.}

\tcfield{Automated by}{\code{RequestLifecycleTest}: ``Draft.Create: withdrawing a Waiting Request cancels its lines \ldots'', ``Draft.Create: a Draft can be withdrawn \ldots'', ``WRONG\_STATE: \ldots an Allocated one cannot be withdrawn'' (Pending part: none yet)}
